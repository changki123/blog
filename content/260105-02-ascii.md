<script>
const video = document.getElementById('video');
const ascii = document.getElementById('ascii');
const resolutionSelect = document.getElementById('resolution');
const styleSelect = document.getElementById('style');
const fpsDisplay = document.getElementById('fps');

const canvas = document.createElement('canvas');
const ctx = canvas.getContext('2d');

// 다양한 ASCII 문자 세트
const charSets = {
  classic: ' .:-=+*#%@',
  block: ' ░▒▓█',
  detailed: ' .\'`^",:;Il!i><~+_-?][}{1)(|\\/tfjrxnuvczXYUJCLQ0OZmwqpdbkhao*#MW&8%B@$',
  matrix: ' .::••●',
};

// 해상도 설정
const resolutions = {
  low: { width: 60, height: 45 },
  medium: { width: 100, height: 75 },
  high: { width: 140, height: 105 },
};

let currentResolution = resolutions.medium;
let currentCharSet = charSets.classic;
let frameCount = 0;
let lastTime = performance.now();

// 초기 캔버스 크기 설정
canvas.width = currentResolution.width;
canvas.height = currentResolution.height;

resolutionSelect.addEventListener('change', (e) => {
  currentResolution = resolutions[e.target.value];
  canvas.width = currentResolution.width;
  canvas.height = currentResolution.height;
  
  const fontSize = e.target.value === 'high' ? 6 : e.target.value === 'medium' ? 8 : 10;
  ascii.style.fontSize = fontSize + 'px';
  ascii.style.lineHeight = fontSize + 'px';
});

styleSelect.addEventListener('change', (e) => {
  currentCharSet = charSets[e.target.value];
  
  if (e.target.value === 'matrix') {
    ascii.style.color = '#0f0';
    ascii.style.textShadow = '0 0 5px #0f0';
  } else {
    ascii.style.color = '#0f0';
    ascii.style.textShadow = 'none';
  }
});

// 비디오 로드되면 자동 시작
video.addEventListener('loadeddata', () => {
  requestAnimationFrame(draw);
});

function draw() {
  if (video.paused || video.ended) return;
  
  // FPS 계산
  frameCount++;
  const now = performance.now();
  if (now - lastTime >= 1000) {
    fpsDisplay.textContent = frameCount;
    frameCount = 0;
    lastTime = now;
  }
  
  ctx.drawImage(video, 0, 0, currentResolution.width, currentResolution.height);
  const imageData = ctx.getImageData(0, 0, currentResolution.width, currentResolution.height);
  
  let asciiImage = '';
  for (let i = 0; i < imageData.data.length; i += 4) {
    const r = imageData.data[i];
    const g = imageData.data[i + 1];
    const b = imageData.data[i + 2];
    const brightness = (r * 0.299 + g * 0.587 + b * 0.114);
    const charIndex = Math.floor((brightness / 255) * (currentCharSet.length - 1));
    asciiImage += currentCharSet[charIndex];
    
    if ((i / 4 + 1) % currentResolution.width === 0) {
      asciiImage += '\n';
    }
  }
  
  ascii.textContent = asciiImage;
  requestAnimationFrame(draw);
}
</script>
